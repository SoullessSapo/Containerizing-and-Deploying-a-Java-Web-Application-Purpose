# Named volumes outlive containers: `docker compose down` keeps the data
$ docker compose down
 Container virtualization-web Removed 
 Container virtualization-db Removed 
 Network virtualization-lab_default Removed 
$ docker compose up -d
 Container virtualization-db Started 
 Container virtualization-web Started 
$ docker compose exec db mongosh workshop --quiet --eval "db.messages.countDocuments()"
1

# `docker compose down -v` also deletes the named volumes, and with them the data
$ docker compose down -v
 Container virtualization-web Removed 
 Container virtualization-db Removed 
 Volume virtualization-lab_mongodb Removed 
 Volume virtualization-lab_mongodb_config Removed 
 Network virtualization-lab_default Removed 
$ docker compose up -d
 Container virtualization-db Started 
 Container virtualization-web Started 
$ docker compose exec db mongosh workshop --quiet --eval "db.messages.countDocuments()"
0
